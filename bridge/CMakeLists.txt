cmake_minimum_required(VERSION 3.20)
project(realm_onnx_cmajor_bridge LANGUAGES CXX)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime SDK prefix")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h
  HINTS "\${ONNXRUNTIME_ROOT}/include" "\${ONNXRUNTIME_ROOT}/include/onnxruntime")
find_library(ONNXRUNTIME_LIBRARY onnxruntime
  HINTS "\${ONNXRUNTIME_ROOT}/lib" "\${ONNXRUNTIME_ROOT}/lib64")
if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
  message(FATAL_ERROR "ONNX Runtime SDK not found; set ONNXRUNTIME_ROOT")
endif()
add_library(realm_onnx_cmajor_bridge SHARED onnx_cmajor_bridge.cpp)
target_compile_features(realm_onnx_cmajor_bridge PRIVATE cxx_std_17)
target_include_directories(realm_onnx_cmajor_bridge PUBLIC \${CMAKE_CURRENT_SOURCE_DIR} \${ONNXRUNTIME_INCLUDE_DIR})
target_link_libraries(realm_onnx_cmajor_bridge PRIVATE \${ONNXRUNTIME_LIBRARY})
target_compile_definitions(realm_onnx_cmajor_bridge PRIVATE REALM_ORT_BRIDGE_BUILD)
if(MSVC)
  target_compile_options(realm_onnx_cmajor_bridge PRIVATE /W4 /permissive-)
else()
  target_compile_options(realm_onnx_cmajor_bridge PRIVATE -Wall -Wextra -Wpedantic)
endif()
